import React, { useState } from 'react';

// قاعدة بيانات أسئلة الأمن السيبراني
const CYBER_QUESTIONS = [
  {
    id: 1,
    question: "ما هو الهجوم الذي يقوم فيه المهاجم باعتراض وتعديل البيانات المرسلة بين طرفين دون علمهما؟",
    options: ["هجوم حجب الخدمة (DDoS)", "رجل في المنتصف (MitM)", "التصيد الاحتيالي (Phishing)", "حقن SQL"],
    correctAnswer: 1
  },
  {

    question: "أي من الخيارات التالية يُعد أفضل ممارسة لإنشاء كلمة مرور قوية؟",

    correctAnswer: 2
  },
  {

    question: "ما هو المصطلح البرمجي الذي يشير إلى برمجيات خبيثة تشفر ملفات المستخدم وتطالب بمقابل مالي لفكها؟",

    correctAnswer: 0
  },

    options: ["التشفير المتماثل", "التحقق الثنائي (MFA)", "جدار الحماية (Firewall)", "بروتوكول HTTPS"],
    correctAnswer: 1
  }
];

// بيانات وهمية لقائمة الأوائل
const LEADERBOARD_DATA = [
  { name: "أحمد المصرى", score: 100, time: "1:45" },
  { name: "سارة المهندس", score: 100, time: "2:10" },
  { name: "عبد الله السيبراني", score: 75, time: "1:30" },
];

export default function CyberExamPlatform() {
  // حالات النظام (States)
  const [step, setStep] = useState('login'); // login, exam, result, leaderboard
  const [studentInfo, setStudentInfo] = useState({ name: '', phone: '', track: 'أساسيات الأمن السيبراني' });
  const [currentQuestionIndex, setCurrentQuestionIndex] = useState(0);
  const [selectedAnswers, setSelectedAnswers] = useState({});
  const [finalScore, setFinalScore] = useState(0);

  // معالجة تسجيل الدخول البدئي
  const handleLogin = (e) => {
    e.preventDefault();
    if (studentInfo.name.trim() && studentInfo.phone.trim()) {
      setStep('exam');
    } else {
      alert("الرجاء إدخال بياناتك كاملة للدخول!");
    }
  };

  // معالجة اختيار الإجابة
  const handleSelectAnswer = (optionIndex) => {
    setSelectedAnswers({
      ...selectedAnswers,
      [currentQuestionIndex]: optionIndex
    });
  };

  // إنهاء الامتحان وحساب الدرجة
  const handleFinishExam = () => {
    let score = 0;
    CYBER_QUESTIONS.forEach((q, index) => {
      if (selectedAnswers[index] === q.correctAnswer) {
        score += 1;
      }
    });
    const percentage = Math.round((score / CYBER_QUESTIONS.length) * 100);
    setFinalScore(percentage);
    setStep('result');
  };

  return (
    <div className="min-h-screen bg-slate-950 text-slate-100 font-sans flex flex-col items-center justify-center p-4 selection:bg-cyan-500 selection:text-slate-950" dir="rtl">
      {/* الخلفية المضيئة المودرن */}
      <div className="absolute top-0 left-1/4 w-96 h-96 bg-cyan-500/10 rounded-full blur-3xl pointer-events-none"></div>
      <div className="absolute bottom-0 right-1/4 w-96 h-96 bg-indigo-500/10 rounded-full blur-3xl pointer-events-none"></div>

      {/* الحاوية الرئيسية للمنصة */}
      <div className="w-full max-w-lg bg-slate-900/80 border border-slate-800 rounded-2xl shadow-2xl backdrop-blur-xl p-6 relative overflow-hidden">
        
        {/* هيدر المنصة الثابت */}
        <div className="text-center mb-6">
          <div className="inline-flex p-3 bg-cyan-950 border border-cyan-800 text-cyan-400 rounded-2xl mb-3 shadow-inner">
            <svg xmlns="http://w3.org" className="h-8 w-8" fill="none" viewBox="0 0 24 24" stroke="currentColor">
              <path strokeLinecap="round" strokeLinejoin="round" strokeWidth={2} d="M12 15v2m-6 4h12a2 2 0 002-2v-6a2 2 0 00-2-2H6a2 2 0 00-2 2v6a2 2 0 002 2zm10-10V7a4 4 0 00-8 0v4h8z" />
            </svg>
          </div>
          <h1 className="text-2xl font-bold bg-gradient-to-r from-cyan-400 to-indigo-400 bg-clip-text text-transparent">منصة درع الأمن السيبراني</h1>
          <p className="text-xs text-slate-400 mt-1">طوّر مهاراتك واختبر جاهزيتك الأمنية الرقمية</p>
        </div>

        {/* 1️⃣ واجهة الدخول والتسجيل */}
        {step === 'login' && (
          <form onSubmit={handleLogin} className="space-y-4">
            <div>
              <label className="block text-sm font-medium text-slate-300 mb-1">المسار الدراسي</label>
              <select 
                className="w-full bg-slate-950 border border-slate-800 rounded-xl px-4 py-3 text-slate-200 focus:outline-none focus:border-cyan-500 transition"
                value={studentInfo.track}
                onChange={(e) => setStudentInfo({...studentInfo, track: e.target.value})}
              >
                <option>أساسيات الأمن السيبراني 🛡️</option>
                <option>اختبار الاختراق الأخلاقي 💻</option>
                <option>التحقيق الجنائي الرقمي 🔍</option>
              </select>
            </div>

            <div>
              <label className="block text-sm font-medium text-slate-300 mb-1">الاسم الرباعي للمشترك</label>
              <input 
                type="text" 
                required
                placeholder="أدخل اسمك الكامل هنا..." 
                className="w-full bg-slate-950 border border-slate-800 rounded-xl px-4 py-3 text-slate-200 placeholder:text-slate-600 focus:outline-none focus:border-cyan-500 transition"
                value={studentInfo.name}
                onChange={(e) => setStudentInfo({...studentInfo, name: e.target.value})}
              />
            </div>

            <div>
              <label className="block text-sm font-medium text-slate-300 mb-1">رقم الهاتف</label>
              <input 
                type="tel" 
                required
                placeholder="01xxxxxxxxx" 
                className="w-full bg-slate-950 border border-slate-800 rounded-xl px-4 py-3 text-slate-200 text-left placeholder:text-slate-600 focus:outline-none focus:border-cyan-500 transition"
                value={studentInfo.phone}
                onChange={(e) => setStudentInfo({...studentInfo, phone: e.target.value})}
              />
            </div>

            <button type="submit" className="w-full bg-gradient-to-r from-cyan-500 to-indigo-600 hover:from-cyan-400 hover:to-indigo-500 text-slate-950 font-bold py-3.5 rounded-xl shadow-lg shadow-cyan-500/20 active:scale-[0.99] transition">
              بدء الاختبار التنافسي 🚀
            </button>

            <div className="relative flex py-2 items-center">
              <div className="flex-grow border-t border-slate-800"></div>
              <span className="flex-shrink mx-4 text-slate-600 text-xs">أو تصفح المنصة</span>
              <div className="flex-grow border-t border-slate-800"></div>
            </div>

            <button type="button" onClick={() => setStep('leaderboard')} className="w-full bg-slate-950 hover:bg-slate-850 border border-slate-800 hover:border-yellow-600/50 text-yellow-500 font-medium py-3 rounded-xl transition flex items-center justify-center gap-2">
              🏆 عرض لوحة المتصدرين والأوائل
            </button>
          </form>
        )}

        {/* 2️⃣ واجهة إجراء الامتحان التفاعلي */}
        {step === 'exam' && (
          <div className="space-y-5">
            <div className="flex justify-between items-center text-sm text-slate-400 border-b border-slate-800 pb-3">
              <span>الممتحن: <strong className="text-slate-200">{studentInfo.name}</strong></span>
              <span className="bg-slate-950 px-2.5 py-1 rounded-md border border-slate-800 text-cyan-400 font-mono">سؤال {currentQuestionIndex + 1} من {CYBER_QUESTIONS.length}</span>
            </div>

            <div className="bg-slate-950 p-4 rounded-xl border border-slate-800 min-h-[80px] flex items-center">
              <p className="text-lg font-medium text-slate-200 leading-relaxed">{CYBER_QUESTIONS[currentQuestionIndex].question}</p>
            </div>

            <div className="space-y-2.5">
              {CYBER_QUESTIONS[currentQuestionIndex].options.map((option, idx) => (
                <button
                  key={idx}
                  onClick={() => handleSelectAnswer(idx)}
                  className={`w-full text-right p-4 rounded-xl border transition-all flex items-center justify-between ${
                    selectedAnswers[currentQuestionIndex] === idx
                      ? 'bg-cyan-950/40 border-cyan-500 text-cyan-300 font-medium shadow-md shadow-cyan-500/5'
                      : 'bg-slate-950 border-slate-800 hover:border-slate-700 text-slate-300'
                  }`}
                >
                  <span>{option}</span>
                  <div className={`w-5 h-5 rounded-full border flex items-center justify-center ${
                    selectedAnswers[currentQuestionIndex] === idx ? 'border-cyan-400 bg-cyan-400' : 'border-slate-700'
                  }`}>
                    {selectedAnswers[currentQuestionIndex] === idx && <div className="w-2 h-2 bg-slate-950 rounded-full"></div>}
                  </div>
                </button>
              ))}
            </div>

            <div className="flex gap-3 pt-3 border-t border-slate-800">
              <button
                disabled={currentQuestionIndex === 0}
                onClick={() => setCurrentQuestionIndex(prev => prev - 1)}
                className="flex-1 bg-slate-950 hover:bg-slate-800 border border-slate-800 disabled:opacity-40 text-slate-300 py-3 rounded-xl transition"
              >
                السابق
              </button>

              {currentQuestionIndex < CYBER_QUESTIONS.length - 1 ? (
                <button
                  onClick={() => setCurrentQuestionIndex(prev => prev 1)}
                  className="flex-1 bg-slate-800 hover:bg-slate-700 border border-slate-700 text-slate-100 py-3 rounded-xl transition"
                >
                  التالي
                </button>
              ) : (
                <button
