    <script>
        let balance = parseInt(localStorage.getItem('cyber_balance')) || 0;
        document.getElementById('money').innerText = balance;
        
        // استعادة الوضع المحفوظ (ليلي/نهاري)
        if (localStorage.getItem('theme') === 'dark') {
            document.documentElement.setAttribute('data-theme', 'dark');
            document.getElementById('themeBtn').innerText = "الوضع النهاري ☀️";
        }

        function toggleTheme() {
            const currentTheme = document.documentElement.getAttribute('data-theme');
            if (currentTheme === 'dark') {
                document.documentElement.removeAttribute('data-theme');
                localStorage.setItem('theme', 'light');
                document.getElementById('themeBtn').innerText = "الوضع الليلي 🌙";
            } else {
                document.documentElement.setAttribute('data-theme', 'dark');
                localStorage.setItem('theme', 'dark');
                document.getElementById('themeBtn').innerText = "الوضع النهاري ☀️";
            }
        }

        function checkAnswer(lessonId, correctAns, reward) {
            const userAns = document.getElementById(`ans${lessonId}`).value.trim();
            if (userAns === correctAns) {
                balance += reward;
                localStorage.setItem('cyber_balance', balance);
                document.getElementById('money').innerText = balance;
                document.getElementById(`btn${lessonId}`).disabled = true;
                document.getElementById(`btn${lessonId}`).innerText = "تم الحل بنجاح! +";
                alert(`أحسنت! الإجابة صحيحة وتم إضافة ${reward}$ إلى رصيدك.`);
            } else {
                alert("إجابة خاطئة! انسخ الكود بدقة وجربه مرة أخرى.");
            }
        }
    </script>
