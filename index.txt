    <script>
        const quizData = [
            {
                question: "ما هو البروتوكول المسؤول عن تشفير اتصالات الويب وصناعة قفل الأمان؟",
                options: ["HTTP", "FTP", "HTTPS", "SSH"],
                correct: 2
            },
            {
                question: "ما نوع الهجوم الذي يتم فيه إغراق الخادم بطلبات وهمية لإيقافه عن العمل؟",
                options: ["Phishing", "DDoS Attack", "SQL Injection", "Man-in-the-Middle"],
                correct: 1
            }
        ];

        let currentQuiz = 0;

        const questionEl = document.getElementById('question');
        const optionsContainer = document.getElementById('options');
        const nextBtn = document.getElementById('next-btn');

        function loadQuiz() {
            deselectAnswers();
            const currentQuizData = quizData[currentQuiz];
            questionEl.innerText = currentQuizData.question;
            optionsContainer.innerHTML = '';

            currentQuizData.options.forEach((option, index) => {
                const button = document.createElement('button');
                button.innerText = option;
                button.className = "w-full text-right p-4 rounded-xl border border-slate-800 bg-slate-900/50 hover:bg-slate-800 hover:border-emerald-500/50 transition duration-200 block";
                button.onclick = () => selectAnswer(index, button);
                optionsContainer.appendChild(button);
            });
        }

        function deselectAnswers() {
            nextBtn.classList.add('hidden');
        }

        function selectAnswer(index, button) {
            const currentQuizData = quizData[currentQuiz];
            const allButtons = optionsContainer.querySelectorAll('button');
            
            allButtons.forEach(btn => btn.disabled = true); // قفل الأزرار بعد الاختيار

            if(index === currentQuizData.correct) {
                button.classList.add('border-emerald-500', 'bg-emerald-950/30', 'text-emerald-400');
            } else {
                button.classList.add('border-red-500', 'bg-red-950/30', 'text-red-400');
                allButtons[currentQuizData.correct].classList.add('border-emerald-500', 'text-emerald-400');
            }
            nextBtn.classList.remove('hidden');
        }

        nextBtn.onclick = () => {
            currentQuiz++;
            if(currentQuiz < quizData.length) {
                loadQuiz();
            } else {
                document.getElementById('quiz-container').innerHTML = `
                    <div class="text-center py-8">
                        <h4 class="text-2xl font-bold text-emerald-400 mb-2">🎉 تهانينا!</h4>
                        <p class="text-slate-400">لقد أتممت الاختبار الأولي بنجاح.</p>
                    </div>
                `;
            }
        };

        loadQuiz();
    </script>
